/**
 * Review panel for actions the assistant proposed.
 *
 * Shows each proposal in full (what it is, where it goes and the exact text)
 * with Approve and Reject buttons. Nothing leaves the machine until Approve is pressed, and
 * an approved action runs exactly once. The list refreshes every five seconds.
 */
export function PendingActions() {
  const [actions, setActions] = useState<PendingAction[]>([]);
  const [notice, setNotice] = useState<string | null>(null);
  const [busy, setBusy] = useState<string | null>(null);
 
  const refresh = useCallback(() => {
    invoke<unknown>("pending_actions")
      .then((reply) => setActions(parsePending(reply)))
      .catch((cause: unknown) => setNotice(t("Could not load proposed actions: {error}", { error: String(cause) })));
  }, []);
 
  useEffect(() => {
    refresh();
    const timer = window.setInterval(refresh, REFRESH_MS);
    return () => window.clearInterval(timer);
  }, [refresh]);
 
  const decide = (id: string, approve: boolean) => {
    setBusy(id);
    invoke<unknown>("decide_action", { id, approve })
      .then((reply) => setNotice(describeOutcome(reply)))
      .catch((cause: unknown) => setNotice(t("Could not complete that: {error}", { error: String(cause) })))
      .finally(() => {
        setBusy(null);
        refresh();
      });
  };
 
  return (
    <section aria-label={t("Proposed actions")} className="mt-6">
      <h2 className="text-lg font-semibold">{t("Proposed actions")}</h2>
      {notice && (
        <p role="status" className="mt-2 text-sm break-words text-zinc-700">
          {notice}
        </p>
      )}
      {actions.length === 0 && <p className="mt-2 text-sm text-zinc-600">{t("Nothing is waiting for you.")}</p>}
      <ul className="mt-2 flex flex-col gap-3">
        {actions.map((action) => (
          <li key={action.id} className="rounded-xl border border-zinc-200 bg-surface p-4">
            <p className="text-xs text-zinc-600">{action.heading}</p>
            {action.title && <h3 className="font-semibold">{action.title}</h3>}
            {action.detail && <p className="text-xs text-zinc-600">{action.detail}</p>}
            {action.body && (
              <pre className="mt-1 max-h-40 overflow-auto text-sm whitespace-pre-wrap">{action.body}</pre>
            )}
            <div className="mt-2 flex gap-2">
              <button
                type="button"
                disabled={busy !== null}
                onClick={() => decide(action.id, true)}
                className="cursor-pointer rounded-md bg-accent px-3 py-1.5 text-sm font-medium text-white hover:bg-accent/90 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-accent disabled:cursor-not-allowed disabled:opacity-50"
              >
                {busy === action.id ? t("Working…") : action.approveLabel}
              </button>
              <button
                type="button"
                disabled={busy !== null}
                onClick={() => decide(action.id, false)}
                className="cursor-pointer rounded-md border border-zinc-400 px-3 py-1.5 text-sm hover:bg-zinc-100 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-accent disabled:cursor-not-allowed disabled:opacity-50"
              >
                {t("Reject")}
              </button>
            </div>

        ))}

    </section>
  );
}
